CREATE TABLE TRANSACTION (
    ID NUMBER(19) GENERATED BY DEFAULT ON NULL AS IDENTITY PRIMARY KEY,
    TRANSACTION_ID VARCHAR2(100),
    AMOUNT NUMBER(15, 2),
    STATUS NUMBER(10),
    BUSSINESS_DATE VARCHAR2(20),
    ACCOUNT_NUMBER VARCHAR2(50),
    VERSION NUMBER(19),
    CREATION_DATE TIMESTAMP
);

CREATE Unique INDEX IDX_TXN_TRANSACTION_ID_1 ON TRANSACTION (TRANSACTION_ID);

CREATE TABLE TRANSACTION_FAIL (
    ID NUMBER(19) GENERATED BY DEFAULT ON NULL AS IDENTITY PRIMARY KEY,
    TRANSACTION_ID VARCHAR2(100),
    AMOUNT NUMBER(15, 2),
    STATUS NUMBER(10),
    BUSSINESS_DATE VARCHAR2(20),
    ACCOUNT_NUMBER VARCHAR2(50),
    FAILURE_REASON VARCHAR2(4000),
    VERSION NUMBER(19),
    CREATION_DATE TIMESTAMP
);

CREATE INDEX IDX_TXN_TRANSACTION_ID_2 ON TRANSACTION_FAIL (TRANSACTION_ID);

CREATE TABLE TRANSACTION_SNDR (
    ID NUMBER(19) GENERATED BY DEFAULT ON NULL AS IDENTITY PRIMARY KEY,
    TRANSACTION_ID VARCHAR2(100),
    VERSION NUMBER(19),
    CREATION_DATE TIMESTAMP
);

CREATE TABLE ERROR_QUEUE_TBL (
    ID NUMBER(19) GENERATED BY DEFAULT ON NULL AS IDENTITY PRIMARY KEY,
    MESSAGE VARCHAR2(4000),
    MESSAGE_HASH VARCHAR2(500),
    VERSION NUMBER(19),
    CREATION_DATE TIMESTAMP
);
