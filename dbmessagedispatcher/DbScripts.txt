CREATE TABLE TRANSACTION_SNDR_FAIL (
    ID NUMBER(19) GENERATED BY DEFAULT ON NULL AS IDENTITY PRIMARY KEY,
    TRANSACTION_ID VARCHAR2(100),
    FAILURE_REASON VARCHAR2(4000),
    VERSION NUMBER(19),
    CREATION_DATE TIMESTAMP
);

CREATE INDEX IDX_TXN_TRANSACTION_ID_3 ON TRANSACTION_SNDR_FAIL (TRANSACTION_ID);
