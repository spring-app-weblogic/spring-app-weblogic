CREATE TABLE EMPLOYEE (
    ID NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    NAME VARCHAR2(100),
    DEPARTMENT VARCHAR2(100),
    status integer
);

CREATE TABLE DEPARTMENT (
    ID NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    NAME VARCHAR2(100)
)

CREATE TABLE FAIL_MSG (
    ID NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    NAME VARCHAR2(100),
    DEPARTMENT VARCHAR2(100),
    FAILUREREASON VARCHAR2(4000)
);
